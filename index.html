// MURO COMUNITARIO BLINDADO Y SIN DUPLICADOS function loadCloudNotes() { try { const saved = localStorage.getItem('enfoque_global_journal'); if (saved) { journalNotes = JSON.parse(saved); } else { journalNotes = [ { id: 1, author: 'Andrés Felipe', date: '3 Oct • 11:30', text: '¡Bienvenidos al muro global de ENFOQUE! A darle con toda.' } ]; } renderJournalNotes(); } catch (e) { journalNotes = [{ id: 1, author: 'Andrés Felipe', date: '3 Oct • 11:30', text: '¡Bienvenidos al muro de ENFOQUE!' }]; renderJournalNotes(); } } function addJournalNote() { const authorInput = document.getElementById('note-author'); const textInput = document.getElementById('note-input'); const author = authorInput.value.trim(); const text = textInput.value.trim(); if(!author || !text) { alert('⚠️ Ingresa tu nombre y mensaje.'); return; } const now = new Date(); const newNote = { id: Date.now(), author: author, date: `${now.toLocaleDateString('es-ES', {month:'short', day:'numeric'})} • ${now.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'})}`, text: text }; // Evitar duplicados consecutivos exactos if(journalNotes.length > 0 && journalNotes[0].text === text && journalNotes[0].author === author) { return; } journalNotes.unshift(newNote); localStorage.setItem('enfoque_global_journal', JSON.stringify(journalNotes)); authorInput.value = ''; textInput.value = ''; renderJournalNotes(); alert('🚀 ¡Comentario publicado con éxito!'); } function renderJournalNotes() { const container = document.getElementById('journal-list'); if(!journalNotes.length) { container.innerHTML = '

Sin comentarios aún.

'; return; } container.innerHTML = journalNotes.map(n => `
${escapeHtml(n.author)} ${n.date}

${escapeHtml(n.text)}

`).join(''); } // Función de seguridad para limpiar textos function escapeHtml(text) { if (!text) return ''; return text.replace(/&/g, "&").replace(//g, ">").replace(/"/g, """).replace(/'/g, "'"); }
